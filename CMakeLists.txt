cmake_minimum_required(VERSION 3.24)
project(SmartCut VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg shared SDK root containing include/, lib/, bin/")
if(NOT FFMPEG_ROOT AND DEFINED ENV{FFMPEG_ROOT})
  set(FFMPEG_ROOT "$ENV{FFMPEG_ROOT}" CACHE PATH "FFmpeg shared SDK root containing include/, lib/, bin/" FORCE)
endif()
if(NOT FFMPEG_ROOT)
  message(FATAL_ERROR
    "Specify the downloaded FFmpeg shared SDK with -DFFMPEG_ROOT=<SDK-root> "
    "or the FFMPEG_ROOT environment variable. The SDK must contain include/, lib/, bin/.")
endif()
get_filename_component(FFMPEG_ROOT "${FFMPEG_ROOT}" ABSOLUTE)
set(FFMPEG_ROOT "${FFMPEG_ROOT}" CACHE PATH "FFmpeg shared SDK root containing include/, lib/, bin/" FORCE)
if(NOT IS_DIRECTORY "${FFMPEG_ROOT}")
  message(FATAL_ERROR "FFMPEG_ROOT is not a directory: ${FFMPEG_ROOT}")
endif()
# Always resolve inside the selected SDK. Cached results from a previous SDK
# must not mix its headers/import libraries with the new runtime DLLs.
foreach(component avformat avcodec avutil)
  unset(${component}_LIBRARY)
  unset(${component}_LIBRARY CACHE)
  find_library(${component}_LIBRARY NAMES ${component}
    PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  if(NOT EXISTS "${FFMPEG_ROOT}/include/lib${component}/${component}.h")
    message(FATAL_ERROR "Missing FFmpeg development header: ${FFMPEG_ROOT}/include/lib${component}/${component}.h")
  endif()
endforeach()
unset(FFMPEG_INCLUDE_DIR)
unset(FFMPEG_INCLUDE_DIR CACHE)
find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h
  PATHS "${FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
message(STATUS "FFmpeg SDK: ${FFMPEG_ROOT}")
add_library(smartcut_core src/platform.cpp src/cli.cpp src/video.cpp src/app.cpp)
target_include_directories(smartcut_core PUBLIC src)
target_include_directories(smartcut_core SYSTEM PUBLIC "${FFMPEG_INCLUDE_DIR}")
target_link_libraries(smartcut_core PUBLIC ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY})
add_executable(smartcut src/main.cpp)
target_link_libraries(smartcut PRIVATE smartcut_core)
if(MSVC)
  target_compile_options(smartcut_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(smartcut PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_definitions(smartcut_core PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(smartcut_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  file(GLOB runtime_dlls "${FFMPEG_ROOT}/bin/*.dll")
  if(NOT runtime_dlls)
    message(FATAL_ERROR "The Windows shared SDK needs runtime DLLs in ${FFMPEG_ROOT}/bin")
  endif()
  if(NOT EXISTS "${FFMPEG_ROOT}/bin/ffmpeg.exe")
    message(FATAL_ERROR "The SDK needs ${FFMPEG_ROOT}/bin/ffmpeg.exe for audio/subtitle processing")
  endif()
  add_custom_command(TARGET smartcut POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${runtime_dlls} "$<TARGET_FILE_DIR:smartcut>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FFMPEG_ROOT}/bin/ffmpeg.exe" "$<TARGET_FILE_DIR:smartcut>"
    COMMAND_EXPAND_LISTS
    VERBATIM)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME cli_help COMMAND smartcut --help)
  add_executable(smartcut_unit_tests tests/unit.cpp)
  target_link_libraries(smartcut_unit_tests PRIVATE smartcut_core)
  add_test(NAME cli_options COMMAND smartcut_unit_tests)
endif()
