cmake_minimum_required(VERSION 3.24)
project(SmartCut VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg shared SDK containing include/, lib/, bin/")
if(NOT FFMPEG_ROOT)
  file(GLOB sdks "${CMAKE_SOURCE_DIR}/deps/ffmpeg-*" "${CMAKE_SOURCE_DIR}/research/h264/deps/ffmpeg-*-win64-gpl-shared-8.1")
  list(LENGTH sdks count)
  if(count EQUAL 1)
    list(GET sdks 0 FFMPEG_ROOT)
  endif()
endif()
foreach(component avformat avcodec avutil)
  find_library(${component}_LIBRARY NAMES ${component} HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
endforeach()
find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
add_library(smartcut_core src/platform.cpp src/cli.cpp src/video.cpp src/app.cpp)
target_include_directories(smartcut_core PUBLIC src)
target_include_directories(smartcut_core SYSTEM PUBLIC "${FFMPEG_INCLUDE_DIR}")
target_link_libraries(smartcut_core PUBLIC ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY})
add_executable(smartcut src/main.cpp)
target_link_libraries(smartcut PRIVATE smartcut_core)
if(MSVC)
  target_compile_options(smartcut_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(smartcut PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_definitions(smartcut_core PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(smartcut_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  file(GLOB runtime_dlls "${FFMPEG_ROOT}/bin/*.dll")
  add_custom_command(TARGET smartcut POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${runtime_dlls} "$<TARGET_FILE_DIR:smartcut>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FFMPEG_ROOT}/bin/ffmpeg.exe" "$<TARGET_FILE_DIR:smartcut>"
    COMMAND_EXPAND_LISTS)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME cli_help COMMAND smartcut --help)
  add_executable(smartcut_unit_tests tests/unit.cpp)
  target_link_libraries(smartcut_unit_tests PRIVATE smartcut_core)
  add_test(NAME cli_options COMMAND smartcut_unit_tests)
endif()
