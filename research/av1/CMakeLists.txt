cmake_minimum_required(VERSION 3.24)
project(av1_smart_render VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg shared SDK root (include/, lib/, bin/)")
if(NOT FFMPEG_ROOT AND DEFINED ENV{FFMPEG_ROOT})
  set(FFMPEG_ROOT "$ENV{FFMPEG_ROOT}" CACHE PATH "FFmpeg shared SDK root" FORCE)
endif()
if(NOT FFMPEG_ROOT)
  message(FATAL_ERROR "Set -DFFMPEG_ROOT=<SDK-root> or the FFMPEG_ROOT environment variable")
endif()
get_filename_component(FFMPEG_ROOT "${FFMPEG_ROOT}" ABSOLUTE)
foreach(component avformat avcodec avutil)
  unset(${component}_LIBRARY CACHE)
  find_library(${component}_LIBRARY NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
unset(FFMPEG_INCLUDE_DIR CACHE)
find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h PATHS "${FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
add_executable(av1_smart_render src/main.cpp)
target_include_directories(av1_smart_render SYSTEM PRIVATE "${FFMPEG_INCLUDE_DIR}")
target_link_libraries(av1_smart_render PRIVATE ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY})
if(MSVC)
  target_compile_options(av1_smart_render PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_definitions(av1_smart_render PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
  file(GLOB runtime_dlls "${FFMPEG_ROOT}/bin/*.dll")
  if(NOT runtime_dlls)
    message(FATAL_ERROR "The selected shared SDK needs bin/*.dll")
  endif()
  add_custom_command(TARGET av1_smart_render POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${runtime_dlls} "$<TARGET_FILE_DIR:av1_smart_render>"
    COMMAND_EXPAND_LISTS VERBATIM)
endif()
enable_testing()
add_test(NAME cli_help COMMAND av1_smart_render --help)
